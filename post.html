<body>

  <main>

    <h1 id="titulo"></h1>

    <p class="meta" id="meta"></p>

    <img id="post-image" src="" alt="Imagen del post" />

    <p class="caption" id="post-creditos"></p>

    <section id="contenido"></section>

  </main>


  <script>

    // Función para obtener parámetros de la URL

    function getQueryParam(param) {

      const urlParams = new URLSearchParams(window.location.search);

      return urlParams.get(param);

    }


    const postId = getQueryParam('id');

    const tituloEl = document.getElementById('titulo');

    const metaEl = document.getElementById('meta');

    const imgEl = document.getElementById('post-image');

    const creditosEl = document.getElementById('post-creditos');

    const contenidoEl = document.getElementById('contenido');


    if (!postId) {

      contenidoEl.innerHTML = '<p class="error">No se especificó ningún post.</p>';

    } else {

      fetch('posts.json')

        .then(response => response.json())

        .then(posts => {

          const post = posts.find(p => p.id === postId);

          if (!post) {

            contenidoEl.innerHTML = '<p class="error">Post no encontrado.</p>';

            return;

          }

          tituloEl.textContent = post.titulo;

          metaEl.textContent = post.fecha + ' | ' + post.categoria;


          if (post.imagen) {

            imgEl.src = 'img/' + post.imagen;

            imgEl.alt = post.titulo;

          } else {

            imgEl.style.display = 'none';

          }


          creditosEl.textContent = post.creditos || '';


          contenidoEl.textContent = post.contenido;

        })

        .catch(() => {

          contenidoEl.innerHTML = '<p class="error">Error al cargar los posts.</p>';

        });

    }

  </script>

</body>
